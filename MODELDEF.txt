// ============================================================================
// RS_VR_Weapons -- WHAT EACH TEST GUN IS DRAWN AS.
//
// GZDoom decides model-vs-sprite from MODELDEF, per class, so each prop class
// has a block here and its card names it (`prop`). The meshes and skins are this
// package's own copies (models/pistols/, models/shotguns/). The shotgun meshes
// are byte-identical to RS_ModelSwapper's re-centred ones as of 2026-09-12.
// Nothing here depends on RS_ModelSwapper.
//
// SCALE AND OFFSET: RS_ModelSwapper's own HUD blocks for these meshes, verbatim
// (its modeldef: the pistols' blocks, MS_AE_Shotgun and MS_Shotgun). An Offset
// belongs to ONE mesh's origin and never transfers to another mesh; where the
// hand holds it is the placement set on top -- the reload system's per-hand seats
// (wm_main / wm_off, its CVARINFO) for the pistols, this package's own sets
// (CVARINFO.txt) for the shotguns.
//
// WHERE THE GRIP LANDS -- AN ESTIMATE, COMPUTED, NOT SEEN IN A HEADSET. On the
// follow-hand path (models.cpp) the placement yaw of -90 turns the mesh, then
// Offset and the placement offsets are added, then it is scaled: the same chain
// RS_ModelSwapper's tools_md3_compensate.py replicates for the HUD path. So
// Offset y runs along the barrel (minus md3 x, times Scale) and z is up. With the
// pistols' seats (the wm_main / wm_off defaults):
//
//   M37A2         its pistol grip (-26.5, 0, -8.5) lands 1.3 map units BEHIND
//                 the M4A3's grip (its magazine centroid), and level with it.
//   Doom shotgun  its wrist (-13, 0, -1.0) lands 0.7 map units behind and 1.4
//                 ABOVE the Pistolet's grip -- a straight stock's wrist is
//                 higher than a pistol grip hangs.
//
// If either sits wrong, the Offset that puts that grip squarely on its pistol's
// is: M37A2 `Offset -0.257 -53.01 -5.625`; Doom shotgun `Offset 0.042 -43.88 -8.05`.
//
// MIRRORED: the M37A2's Scale x is -1.1, as the M4A3's is -0.82. On this path the
// mesh is turned before it is scaled, so x scale acts ACROSS the gun: mirrored
// left for right, never front for back. NOAUTOREVERSE on every gun: a gun in your
// left hand is still that gun, not its mirror image.
//
// FrameIndex WMPR: the reload system's one-pixel prop sprite, which every
// WM_Prop's Spawn state shows.
// ============================================================================

// RS_GRENADE'S MODELDEF, folded in 09-14 from presidentkoopa/RS_Grenade ffff0c6 -- copied whole, every name unchanged so saved values and bindings carry over. Loading RS_Grenade.pk3 as well is a startup error.

// THE GRENADE MODEL, from RS_Main.
//
// ONE POSE ON EVERY FRAME, and that is RS_Main's finding carried over rather
// than rediscovered: a sprite-frame cycle does nothing to a model, so the
// tumble has to be REAL rotation. RS_VRGrenadeThrown.Tick turns roll and pitch
// a few degrees a tic once it leaves your hand, and the model follows because
// of the two USEACTOR flags below.
//
// A Skin line, because its absence is silent: the model renderers skip a mesh
// outright when no skin resolves -- no warning, no error, no draw.
//
// ONE REAL SIZE EVERYWHERE (fixed 2026-09-14). In your hand, in the air and on
// the floor the grenade is the same 17.0 mesh units tall at 3.26 map units, a
// ~0.1 m grenade: Scale 0.5643 on the hand and weapon paths (1 model unit = 0.34
// map units there), Scale 0.1919 (= 0.5643 x 0.34) on the world path (1 model
// unit = 1 map unit). The thrown grenade used to be 5.1 map units and the held
// one 0.03 (see the held blocks).
//
// SAFE IS FRAME 2, NOT FRAME 0. nade.md3's frame 0 is the start of a
// first-person raise, the body tilted 45 degrees and twenty units away from
// where it stands; frames 2-10 and 16 hold the body upright. Frame 2 is the
// grenade whole and standing -- pin in, lever on -- so safe and live (16, lever
// flown) share one attitude, and going from one to the other no longer turns
// the grenade 45 degrees.
//
// A PIVOT PER FRAME. PivotOffset is part of the state a FrameIndex line copies,
// so each frame below sets its own just before its FrameIndex. Frame 16's body
// sits 4 mesh units higher than frame 2's, so one pivot for both put one of
// them off its actor -- that was the ~10 map units the live grenade flew beside
// its own position, with the old frame-0 pivot. Measured out of nade.md3 (body =
// low_lowbodymesh + the fuze bushing), each value is that frame's point times
// the block's Scale.

Model RS_VRGrenadeThrown
{
	Path "models/grenade"
	Model 0 "nade.md3"
	Skin 0 "nade.png"

	// A SECOND COPY OF THE MESH, WEARING A RED SKIN.
	//
	// This is how the fuse is shown, and it is a skin swap rather than anything
	// cleverer because the clever options do not work here. A MODELDEF
	// Translation is a palette remap and this mesh is skinned with a truecolor
	// PNG, so a translation does nothing at all. RenderStyle Stencil works, but
	// only on the world path: RenderHUDModel passes playermo->RenderStyle
	// (models.cpp:1408) -- the PLAYER's -- so stencilling a psprite-drawn model
	// would tint the whole player instead of the grenade.
	//
	// A second model index with its own skin is immune to all of that. It is the
	// same geometry addressed by a different FrameIndex, so switching to it costs
	// one frame number and works identically on both draw paths.
	Model 1 "nade.md3"
	Skin 1 "nade_red.png"
	Scale 0.1919 0.1919 0.1919

	// PIVOTOFFSET, BECAUSE THE MESH IS NOT CENTRED ON ITS OWN ORIGIN.
	//
	// A Doom actor rests with its origin ON the floor, so the pivot is the body's
	// BASE centre: drawn raw the grenade sat underground and off to one side of
	// where it actually was.
	//
	// PIVOTOFFSET AND NOT OFFSET, and the difference is the whole reason the
	// engine gained the field. Offset is applied AFTER the rotations, so it moves
	// the model without moving the point the model turns about -- it would have
	// put the grenade on the floor and left it ORBITING its old origin every time
	// it tumbled. PivotOffset is subtracted BEFORE them, which is the ordinary
	// v' = R * (v - p): the mesh is moved onto its intended centre of rotation
	// first, then spun about it.

	// FLAGS BEFORE THE FrameIndex LINES. Each FrameIndex pushes a COPY of the
	// in-progress frame carrying whatever flags are set at that moment, so a
	// flag written after the last one parses cleanly and applies to nothing.
	NoInterpolation
	USEACTORPITCH
	USEACTORROLL

	// THE FRAME TABLE, AND THE ORDER OF THESE LINES IS LOAD-BEARING.
	//
	// A FrameIndex line sets the frame for the ONE model index it names and
	// leaves every other index at whatever the previous line left it -- then
	// pushes a COPY of that accumulated state. So writing the plain frames and
	// the red frames as two independent groups does not give you two variants:
	// it gives you entries where BOTH models are set, and both meshes draw, one
	// inside the other. The red one wins because it is drawn second, which is
	// why the grenade came out red permanently rather than flashing.
	//
	// The carry-over is also the fix. A model index parked at -1 is not drawn, so
	// setting the unwanted one to -1 FIRST makes every line after it clean until
	// something changes it back.
	//
	// H and I are scratch: their only job is to carry that -1 into the lines
	// below them. They are never displayed -- nothing sets frame H or I -- and
	// the sprite lookup only validates the four-letter name, not the letter, so
	// they cost two unused table entries and nothing else. E mirrors A and F
	// mirrors G, so every state has a pose that already exists and nothing is
	// faked; eight sprite letters is all JGRN has, which is exactly enough.

	FrameIndex JGRN H 1 -1    // from here down: the red mesh is off
	PivotOffset 5.4512 0.0 -1.8680    // frame 2's base centre (28.406, -0.008, -9.734) x 0.1919
	FrameIndex JGRN A 0 2     // safe
	PivotOffset 5.4512 0.0 -1.1004    // frame 16's base centre (28.406, -0.008, -5.734) x 0.1919
	FrameIndex JGRN G 0 16    // live -- lever flown

	FrameIndex JGRN I 0 -1    // from here down: the plain mesh is off
	PivotOffset 5.4512 0.0 -1.8680
	FrameIndex JGRN E 1 2     // safe, red
	PivotOffset 5.4512 0.0 -1.1004
	FrameIndex JGRN F 1 16    // live, red
}
