// PLACEMENT SETS, read by the RENDERER every frame it draws (MODELDEF
// PlacementCVars on each prop), so a slider moves the gun while the menu is
// open. All ten names per set, sizes above zero: the renderer ignores a zero
// or negative _scale.
//
// SEEDED WITH THE PISTOLS' OWN SEATS: wm_main's defaults for the main-hand
// gun, wm_off's for the off-hand gun. A seat belongs to the HAND and the
// MODELDEF Offset to the MESH, so the moment a shotgun draws on its own set it
// sits exactly where it sits on the pistol's set today.
//
// LIVE: the reload system used to force every prop onto the pistol's prefix
// (rig.zs EnsureProp); that was removed in e896d31, so each shotgun now draws
// on its own set below.
//
// NEW NAMES, NOT SHARED WITH THE RELOAD SYSTEM: a cvar declared twice in the
// load order is a conflict, and a saved value outranks a changed default.
// ============================================================================

// RS_Grenade settings.
//
// SERVER OR USER, AND WHY IT MATTERS (netplay, 2026-09-14). A setting that DECIDES
// THE GAME -- whether the grenade works at all, whether a pull lights the fuse,
// how long the fuse burns, how it flies, bounces and settles, whether it is live
// or a dud, when a dud becomes a pickup, whether you start with some -- is
// `server`: one value, the host's, the same on every machine. As `user` values
// they were read through the console player on each machine, so two players
// with different settings saw one grenade do two different things.
// CVar.GetCVar(name, player) hands back a player's own copy only for userinfo
// (user) cvars; a server cvar returns the one synced value, so the reads in the
// code stay as they are.
//
// A setting that only changes what YOU see or feel -- seating, drawing, flashes,
// light, tumble, debug, and the pin gestures' own reach and hold, which are judged
// on the thrower's machine alone -- stays `user`.
//
// Moving a setting from user to server leaves a value you saved as `user` behind:
// the server copy starts from the default below.

server bool  rsvg_enable = true;

// WHEN THE FUSE LIGHTS. The one decision that sets the whole feel.
//   true  -- the PIN starts it. Holding a live grenade is dangerous, cooking
//            works, and fumbling drops it lit at your feet.
//   false -- the LEVER starts it, as a real grenade does. Safe, no tension.
// OFF BY DEFAULT, for now. The fuse starts when the lever flies rather than
// when the pin comes out, so nothing can detonate in your hand.
//
// Cooking is the better feature and this should go back to true once the pin
// gestures are trustworthy -- but a default that can kill you every three
// seconds because an unrelated gesture misfired is not a default, it is a trap.
server bool  rsvg_cook = true;
server float rsvg_fuse = 3.0;

// NO PIN-REACH CVAR. How close the free hand must be is the OVAL's question,
// answered by RS_Reach.ScoreAt against the volume rs_grab_o_scale_* sizes and
// RS_GrabViz draws. A number here would be a second opinion about the same
// thing, and the two would drift.

// THE FACE ROUTE. Two radii, not one -- see the note in the code. The hand has
// to have been clear of your head before coming to it, so resting it near your
// face does nothing.
user float rsvg_face_near = 16.0;
user float rsvg_face_far  = 24.0;
user bool  rsvg_face = true;

// Ships OFF, matching every other subsystem trace in this family.
//
// IT DID NOT. Found while touching this file for the throw upgrade below --
// the default here was `true`, directly contradicting its own comment, which
// means every install has been printing the alive/gesture/face traces to
// console every second since the day this shipped. Corrected, and see
// rsvg_cfgver: an existing ini already carrying `1` from playing keeps that
// value untouched, so this reaches only fresh installs on its own -- anyone
// already running it explicitly turns it off once, same as any other
// hand-tuned setting.
user bool  rsvg_debug = false;

// DIAGNOSTIC -- the weapon slot table at every spawn, off.
//
// It used to print eleven lines per spawn with no switch at all, ten of them the word "empty",
// wearing the WM tag on the same console the reload system uses -- so it read as somebody else's
// noise while being ours. On, it now lists only the slots that hold something and counts the rest
// in one line.
user bool wm_slot_debug = false;

server bool rsvg_start = true;

// SEATING THE HELD GRENADE. Metres, like every placement value downstream of a
// Follow flag -- the menu steps these by 0.01 (a centimetre) for that reason.
user float rsvg_hm_ofs_x = 0.0;
user float rsvg_hm_ofs_y = 0.0;
user float rsvg_hm_ofs_z = 0.0;
user float rsvg_hm_yaw = 0.0;
user float rsvg_hm_pitch = 0.0;
user float rsvg_hm_roll = 0.0;
user float rsvg_hm_scale = 1.0;

user float rsvg_ho_ofs_x = 0.0;
user float rsvg_ho_ofs_y = 0.0;
user float rsvg_ho_ofs_z = 0.0;
user float rsvg_ho_yaw = 0.0;
user float rsvg_ho_pitch = 0.0;
user float rsvg_ho_roll = 0.0;
user float rsvg_ho_scale = 1.0;

// THE FLIGHT. All four are judged by throwing it, so all four are live.
// gravity 1.0 is Doom normal; below that floats, above that drops hard.
// bounce/wallbounce are how much speed survives a hit -- 0 is a dead thud,
// 1 would never settle. Doom's own rocket-ish default sits near 0.3.
// throw is a grenade-only trim ON TOP of the shared rs_throw_scale.
// GRAVITY 0.27, NOT 0.7, AND NOT 1.0.
//
// Doom gravity is not earth gravity. One map unit per tic per tic works out to
// roughly 36 m/s^2 -- about 2.7 times real -- which is why everything in Doom
// falls like it is made of lead and why a thrown object drops out of the air
// almost as soon as it leaves your hand.
//
// 0.7 was inherited from RS_Main's grenade, where it is right: that one is a
// PROJECTILE launched at a fixed 10-60 speed, so a heavy fall is what keeps it
// from sailing. This one leaves at whatever your arm actually did -- around 8
// m/s for a normal throw -- and under 2.7x gravity that is a descending line
// rather than a toss.
//
// 1 / 2.7 = 0.37 would be exactly earth. 0.27 is a little lighter still,
// because the throw also has to cover Doom's rooms, which are larger than they
// look. Tune it -- but start well under 0.5, not near 1.
server float rsvg_gravity    = 0.27;
server float rsvg_bounce     = 0.5;
server float rsvg_wallbounce = 0.25;
server float rsvg_throw      = 1.0;

// THE AIMED THROW'S SPEED, in map units per tic -- the desktop co-op route (rsvg-throwaim), where there
// is no hand motion to measure. 22 is about what a firm VR throw measures. SERVER, because it decides
// where a grenade lands and every machine derives the same vector from the pawn's own aim.
server float rsvg_aimthrow   = 22.0;

// THE BLAST, SCALED. Both rings together (RSVG_Warhead): 90 inside 96 units with the first 24 at full
// strength, and a 35 concussion gone by 192. It was 85 out to 200 PLUS 75 out to 255 -- 160 damage
// across nearly eight metres, where Doom's own rocket is 128 at 128 -- so you stood inside your own
// grenade unless you threw it across a room, and Black Metal multiplies what you take by 1.5.
server float rsvg_blast      = 1.0;

// WHEN A RELEASE IS A DROP AND NOT A THROW, in map units per tic of hand motion (one unit a tic is
// very nearly one metre a second). Below this, opening your hand puts the grenade AT YOUR FEET
// rather than launching it -- the owner's call, 09-18. A tap on the trigger used to cost you a
// grenade, and this is the same line RS_Throw already draws for everything else that leaves a hand
// (rs_throw_min); it is only consulted when RS_WorldHands is absent and the grenade is measuring
// its own swing.
server float rsvg_throw_min  = 1.17;


// THE ARC. A fraction of your throw's horizontal speed added as lift, so it
// scales with how hard you threw instead of arcing every throw identically.
// 0 is dead flat -- whatever your arm did and nothing else.
server float rsvg_lift = 0.35;

// How close the opposite hand has to be to take the pin, palm to palm, in map
// units. 20 is about 60cm -- deliberately generous, because this is a two-hand
// act and not a precision test.
user float rsvg_pin_reach = 20.0;

// Bumped by the code when a default changes for a reason a player should not
// have to know about. A new cvar has no saved value, so this reads 0 exactly
// once per install -- which is the only way a corrected default can reach
// somebody who has already played.
user int rsvg_cfgver = 0;

// GROUND FRICTION. Doom gives a thrown actor none of its own, so without this a
// grenade slides until it hits a wall. 1.0 is frictionless ice, lower stops it
// sooner.
server float rsvg_friction = 0.82;

// Tumble in the air, degrees per tic. 0 is dead still -- Brutal Doom's grenade
// does not rotate its model at all and reads fine, so this is taste. This is
// a FLAT BASELINE that always applies -- see rsvg_spin_wrist below for the
// part that comes from how you actually threw it.
user float rsvg_spin = 3.5;

// HOW MUCH OF YOUR OWN WRIST SNAP RIDES ON TOP OF THE BASELINE ABOVE.
// RS_ThrowService already measures this (RS_ShieldSaw's spin comes from the
// same request) -- the grenade just never asked before. 0 disables it and
// every throw tumbles identically again, exactly as before this existed. 1.0
// adds your measured wrist rate degree-for-degree; higher exaggerates it.
// Silently 0 in effect if RS_WorldHands (or RS_ThrowService) is not loaded --
// there is no local fallback for spin, only for the throw's velocity.
user float rsvg_spin_wrist = 1.0;

// HOW THE HELD GRENADE IS DRAWN. Nothing about the gestures changes -- the pin
// and the throw read real palm and HMD positions either way.
//   false -- world actor. Occluded by walls, lit by the room. Its own seating.
//   true  -- psprite, like every gun. Draws over the world and is lit by the
//            HUD path, but needs no second actor.
user bool rsvg_psprite = false;

// Seating for the PSPRITE route. Separate from rsvg_hm/rsvg_ho because the two
// paths are a factor of 100 apart in scale and are not interchangeable.
user float rsvg_ps_ofs_x = 0.0;
user float rsvg_ps_ofs_y = 0.0;
user float rsvg_ps_ofs_z = 0.0;
user float rsvg_ps_yaw = 0.0;
user float rsvg_ps_pitch = 0.0;
user float rsvg_ps_roll = 0.0;
user float rsvg_ps_scale = 1.0;

// A THROWN DUD BECOMES A PICKUP when it is at least this old AND a player is this
// close. delay is in tics (35 = one second) and exists so it cannot convert on
// the way out of your hand; range is in map units (34 = one metre).
server float rsvg_dud_delay = 35.0;
server float rsvg_dud_range = 96.0;

// HOW LONG THE GRENADE MUST STAY AT YOUR FACE, in tics (35 = one second).
// This is what separates bringing it to your mouth from swinging past your head
// on the way to a throw -- without it, every throw pulled its own pin.
user float rsvg_face_hold = 12.0;

// ARMING. Off (default): the pin has to come out first -- the other hand's
// fire within reach, or a dwell at your face -- and a throw with the pin still
// in is a dud that settles and turns back into a pickup. On: throwing is the
// arming; every throw is live and the pin is decorative.
server bool rsvg_autoarm = false;

// THE FUSE, SEEN. Red / neutral / red on the grenade itself, accelerating as
// it burns down. Same information the ticking carries, in the channel you cannot
// lose in a firefight.
user bool rsvg_flash = true;

// THE OPPOSITE-HAND PIN PULL. Off by default: two hands and a small object at
// arm's length is a finicky gesture, and there is no reach value that is both
// reliable to hit and tight enough not to misfire. The face route needs one hand
// and a deliberate hold, and does the same job better.
user bool rsvg_offhand_pin = false;

// The room-lighting glow. OFF: it worked too well -- bright enough to read on
// the grenade also meant lighting the whole room through it, which with a
// darkness mod is a flashbang in your hand every three seconds. The red skin
// says the same thing and stays on the object it describes. Kept because in a
// genuinely black room it is the only thing that shows where a thrown one went.
user bool rsvg_light = false;

// IMPACT OR TIMER.
//   false -- it bounces and the fuse decides. Bank one off a wall, roll it
//            round a corner, cook one and have it go off in the air.
//   true  -- the first thing it touches ends it. Nothing comes back at you
//            off a doorframe, and cooking stops mattering.
// Read at spawn, not live: a grenade that changed its mind mid-flight
// because a slider moved would be genuinely unpredictable.
server bool rsvg_impact = false;
